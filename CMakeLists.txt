cmake_minimum_required(VERSION 3.20)
project(LanternHall LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(SDL3 CONFIG REQUIRED)
add_executable(lantern-hall src/app.c src/draw.c src/hall.c src/scene.c)
target_link_libraries(lantern-hall PRIVATE SDL3::SDL3)
if(NOT MSVC)
  target_link_libraries(lantern-hall PRIVATE m)
endif()
if(WIN32)
  add_custom_command(TARGET lantern-hall POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3-shared> $<TARGET_FILE_DIR:lantern-hall>)
endif()
enable_testing()
add_executable(unit-hall tests/unit.c src/hall.c src/scene.c)
target_include_directories(unit-hall PRIVATE src)
if(NOT MSVC)
  target_link_libraries(unit-hall PRIVATE m)
endif()
add_test(NAME hall-rules-and-scenes COMMAND unit-hall)

add_executable(replay-hall tests/replay.c src/hall.c)
target_include_directories(replay-hall PRIVATE src)
if(NOT MSVC)
  target_link_libraries(replay-hall PRIVATE m)
endif()
add_test(NAME solo-and-split-completions COMMAND replay-hall)
