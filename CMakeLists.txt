cmake_minimum_required(VERSION 3.20)
project(LanternHall LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(SDL3 CONFIG REQUIRED)
set(LANTERN_SOURCES src/app.c src/draw.c src/hall.c src/campaign.c src/scene.c)
add_executable(lantern-hall ${LANTERN_SOURCES})
add_library(lantern_native SHARED ${LANTERN_SOURCES})
set_target_properties(lantern_native PROPERTIES PREFIX "" WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_link_libraries(lantern_native PRIVATE SDL3::SDL3)
if(NOT MSVC)
  target_link_libraries(lantern_native PRIVATE m)
endif()
add_custom_command(TARGET lantern_native POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  ${CMAKE_CURRENT_SOURCE_DIR}/scripts/native_peer.py $<TARGET_FILE_DIR:lantern_native>/native_peer.py)
target_link_libraries(lantern-hall PRIVATE SDL3::SDL3)
if(NOT MSVC)
  target_link_libraries(lantern-hall PRIVATE m)
endif()
if(WIN32)
  add_custom_command(TARGET lantern-hall POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3-shared> $<TARGET_FILE_DIR:lantern-hall>)
endif()
enable_testing()
add_executable(unit-hall tests/unit.c src/hall.c src/campaign.c src/scene.c)
target_include_directories(unit-hall PRIVATE src)
if(NOT MSVC)
  target_link_libraries(unit-hall PRIVATE m)
endif()
add_test(NAME hall-rules-and-scenes COMMAND unit-hall)

add_executable(replay-hall tests/replay.c src/hall.c src/campaign.c)
target_include_directories(replay-hall PRIVATE src)
if(NOT MSVC)
  target_link_libraries(replay-hall PRIVATE m)
endif()
add_test(NAME solo-and-split-completions COMMAND replay-hall)
